cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
 set(CMAKE_CUDA_ARCHITECTURES 120)
endif()
project(awe_blas VERSION 0.4.0 LANGUAGES C CXX CUDA)
find_package(CUDAToolkit 13.2 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
 set(CMAKE_CUDA_ARCHITECTURES 120)
endif()
option(AWE_ENABLE_OZAKI2 "Build the optional Ozaki II backend" ON)
add_library(awe_blas SHARED src/awe_blas.cu)
if(AWE_ENABLE_OZAKI2 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/ozaki2_backend.cu")
 target_sources(awe_blas PRIVATE src/ozaki2_backend.cu)
 target_compile_definitions(awe_blas PRIVATE AWE_HAS_OZAKI2=1)
endif()
target_include_directories(awe_blas PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(awe_blas PRIVATE CUDA::cublasLt PUBLIC CUDA::cudart)
set_target_properties(awe_blas PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
include(GNUInstallDirs)
install(TARGETS awe_blas EXPORT awe_blasTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/awe_blas.h include/awe_blas.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT awe_blasTargets NAMESPACE awe:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/awe_blas)
option(AWE_BUILD_TESTS "Build API correctness tests" ON)
if(AWE_BUILD_TESTS)
 enable_testing()
 add_executable(awe_smoke tests/smoke.cu)
 target_link_libraries(awe_smoke PRIVATE awe_blas)
 add_test(NAME awe_smoke COMMAND awe_smoke)
 add_executable(awe_fp64_contract tests/fp64_contract.c)
 target_link_libraries(awe_fp64_contract PRIVATE awe_blas)
 add_test(NAME awe_fp64_contract COMMAND awe_fp64_contract)
 add_executable(awe_crt_smoke tests/crt_smoke.cu)
 target_link_libraries(awe_crt_smoke PRIVATE awe_blas)
 add_test(NAME awe_crt_smoke COMMAND awe_crt_smoke)
endif()

option(AWE_BUILD_BENCHMARKS "Build same-device integer benchmarks" ON)
if(AWE_BUILD_BENCHMARKS)
 add_executable(awe_bench_i8 bench/bench_i8.cu)
 target_link_libraries(awe_bench_i8 PRIVATE awe_blas CUDA::cublas CUDA::cublasLt)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(awe_blasConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/awe_blasConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/awe_blas)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/awe_blasConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/awe_blasConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/awe_blasConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/awe_blas)
set(AWE_GEMMUL8_ROOT "" CACHE PATH "Optional pinned GEMMul8 source and static objects for comparison")
if(AWE_BUILD_BENCHMARKS AND AWE_GEMMUL8_ROOT)
 add_executable(awe_compare_i8 bench/compare_i8.cu)
 target_include_directories(awe_compare_i8 PRIVATE ${AWE_GEMMUL8_ROOT}/include)
 target_link_libraries(awe_compare_i8 PRIVATE awe_blas CUDA::cublas CUDA::cublasLt ${AWE_GEMMUL8_ROOT}/gemm_double.o ${AWE_GEMMUL8_ROOT}/libgemmul8.a)
endif()

option(AWE_BUILD_EXAMPLES "Build the C and C++ examples" ON)
if(AWE_BUILD_EXAMPLES)
 add_executable(awe_example_c examples/minimal.c)
 target_link_libraries(awe_example_c PRIVATE awe_blas)
 add_executable(awe_example_cpp examples/minimal.cpp)
 target_link_libraries(awe_example_cpp PRIVATE awe_blas)
endif()
install(DIRECTORY docs examples DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/awe_blas)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/awe_blas)
